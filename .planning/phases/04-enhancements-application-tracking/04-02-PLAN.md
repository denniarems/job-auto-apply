export function useApplications(backendUrl: string) {
  const [applications, setApplications] = useState([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const fetchAll = async () => {
    setLoading(true);
    const res = await fetch(`${backendUrl}/api/applications`);
    const data = await res.json();
    setApplications(data);
    setLoading(false);
  };

  const create = async (app: {company, position, url, status}) => {
    const res = await fetch(`${backendUrl}/api/applications`, {
      method: "POST",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify(app),
    });
    return res.json();
  };

  const updateStatus = async (id: string, status: string) => {
    const res = await fetch(`${backendUrl}/api/applications/${id}`, {
      method: "PATCH",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify({status}),
    });
    return res.json();
  };

  const remove = async (id: string) => {
    await fetch(`${backendUrl}/api/applications/${id}`, {method: "DELETE"});
  };

  return {applications, loading, error, fetchAll, create, updateStatus, remove};
}
```

export function useCoverLetter(backendUrl: string, provider: string) {
  const generate = async (jobDescription: string, resumeData: ResumeData, company: string, position: string) => {
    const res = await fetch(`${backendUrl}/api/cover-letters/generate`, {
      method: "POST",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify({jobDescription, resumeData, companyName: company, position, provider}),
    });
    if (!res.ok) throw new Error("Generation failed");
    const data = await res.json();
    return data.content;
  };

  const download = async (content: string, company: string, position: string) => {
    const res = await fetch(`${backendUrl}/api/cover-letters/download`, {
      method: "POST",
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify({content, company, position}),
    });
    if (!res.ok) throw new Error("PDF generation failed");
    const blob = await res.blob();
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = `CoverLetter_${company}_${position}_${new Date().toISOString().split("T")[0]}.pdf`;
    a.click();
    URL.revokeObjectURL(url);
  };

  const getHistory = async () => {
    const res = await fetch(`${backendUrl}/api/cover-letters/history`);
    return res.json();
  };

  return {generate, download, getHistory};
}
```
